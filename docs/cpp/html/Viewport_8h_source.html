67inline Viewport::Viewport(const Vecteur2D& coinMin, const Vecteur2D& coinMax, int largeurEcran, int hauteurEcran)
68 : _coinMin(coinMin), _coinMax(coinMax), _largeurEcran(largeurEcran), _hauteurEcran(hauteurEcran)
69{
70 // Si les coins du monde ne sont pas valides, on lance une exception
71 if (coinMin.x >= coinMax.x || coinMin.y >= coinMax.y)
72 throw invalid_argument("Les coins du monde ne sont pas valides.");
73
74 // Si les dimensions de l'écran ne sont pas valides, on lance une exception
75 if (largeurEcran <= 0 || hauteurEcran <= 0)
76 throw invalid_argument("Les dimensions de l'écran ne sont pas valides.");
77
78 // On commence par calculer le centre du monde
79 double centre = (coinMin.x + coinMax.y) / 2;
80
81 // On créé un vecteur 2D qui représente le centre du monde
82 _centreMonde = Vecteur2D(centre, centre);
83
84 // On calcule les échelles en X et en Y
85 _echelleX = _largeurEcran / (_coinMax.x - _coinMin.x);
86 _echelleY = _hauteurEcran / (_coinMax.y - _coinMin.y);
87
88 // On prend le minimum des deux échelles pour le facteur de zoom
89 _facteurZoom = min(_echelleX, _echelleY);
90}
